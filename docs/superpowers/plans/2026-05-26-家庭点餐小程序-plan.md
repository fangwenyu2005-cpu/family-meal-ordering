```prisma
model SpecGroup {
  id         Int          @id @default(autoincrement())
  name       String       @db.VarChar(50)
  isRequired Boolean      @default(true) @map("is_required")
  options    SpecOption[]

}

model SpecOption {
  id         Int           @id @default(autoincrement())
  groupId    Int           @map("group_id")

  priceDelta Decimal       @default(0) @map("price_delta") @db.Decimal(10, 2)

  @@map("spec_options")
}

model ProductSpec {
  productId   Int       @map("product_id")

  product     Product   @relation(fields: [productId], references: [id])

  createdAt   DateTime @default(now()) @map("created_at")

  @@map("tables")
```

```js
const jwt = require('jsonwebtoken');
const { jwtSecret } = require('../config');

function authRequired(req, res, next) {
  const header = req.headers.authorization;
  if (!header || !header.startsWith('Bearer ')) {
    return res.status(401).json({ error: '未登录' });
  }
  try {
    req.user = jwt.verify(header.slice(7), jwtSecret);
    next();
  } catch {
    return res.status(401).json({ error: '登录已过期' });
  }
}

module.exports = { authRequired };
```

### Task 1.3: 订单 API（公开 + 顾客端）

**Files:**
- Create: `server/src/routes/orders.js`

- [ ] **Step 1: 提交订单 `POST /api/orders`**

```js
router.post('/', async (req, res, next) => {
  try {
    const { tableId, dineType, items } = req.body;

    // 计算总价
    let totalPrice = 0;
    const orderItemsData = [];

    for (const item of items) {
      const product = await req.prisma.product.findUnique({
        where: { id: item.productId },
      });
      if (!product) {
        return res.status(400).json({ error: `商品 ${item.productId} 不存在` });
      }

      // 计算规格加价
      let itemPrice = product.basePrice;
      if (item.specOptionIds && item.specOptionIds.length > 0) {
        const specOptions = await req.prisma.specOption.findMany({
          where: { id: { in: item.specOptionIds } },
        });
        for (const opt of specOptions) {
          itemPrice = itemPrice.plus(opt.priceDelta);
        }
      }

      totalPrice = totalPrice + itemPrice * item.quantity;

      orderItemsData.push({
        productId: item.productId,
        quantity: item.quantity,
        price: itemPrice,
        specOptionIds: item.specOptionIds || [],
      });
    }

    // 创建订单（事务）
    const order = await req.prisma.$transaction(async (tx) => {
      const newOrder = await tx.order.create({
        data: {
          tableId,
          dineType,
          totalPrice,
          items: {
            create: orderItemsData.map((oi) => ({
              productId: oi.productId,
              quantity: oi.quantity,
              price: oi.price,
              specs: {
                create: oi.specOptionIds.map((specId) => ({
                  specOptionId: specId,
                })),
              },
            })),
          },
        },
        include: {
          items: {
            include: {
              product: true,
              specs: { include: { specOption: true } },
            },
          },
        },
      });
      return newOrder;
    });

    res.status(201).json(order);
  } catch (err) {
    next(err);
  }
});
```

```js
      req.prisma.order.count({

      req.prisma.order.count({
        where: { status: { not: 'CANCELLED' } },
      }),
```

```js
      JOIN products p ON oi.product_id = p.id
      JOIN categories c ON p.category_id = c.id
      JOIN orders o ON oi.order_id = o.id
      WHERE o.status != 'CANCELLED'
      GROUP BY p.id, p.name, c.name
      ORDER BY total_qty DESC
      LIMIT 20
    `;
    res.json(ranking);
  } catch (err) {
    next(err);
  }
});
```

```js
App({
  globalData: {
    tableId: null,
    tableNumber: '',
    apiBase: 'https://your-api-url', // 后续替换为实际域名
    cart: [],
  },

  onLaunch(options) {
    // 解析二维码参数获取桌号
    if (options.query && options.query.table) {
      this.globalData.tableNumber = options.query.table;
      this.verifyTable(options.query.table);
    }
  },

  verifyTable(code) {
    wx.request({
      url: `${this.globalData.apiBase}/api/tables/${code}`,
      success: (res) => {
        if (res.statusCode === 200) {
          this.globalData.tableId = res.data.id;
          this.globalData.tableNumber = res.data.tableNumber;
        }
      },
    });
  },

  addToCart(product, specOptions, quantity) {
    const cart = this.globalData.cart;
    const key = `${product.id}_${specOptions.map(s => s.id).sort().join('_')}`;
    const existing = cart.find(item => item.key === key);
    if (existing) {
      existing.quantity += quantity;
    } else {
      cart.push({ key, product, specOptions, quantity });
    }
  },

  getCartTotal() {
    let total = 0;
    let count = 0;
    this.globalData.cart.forEach(item => {
      const itemPrice = item.product.basePrice +
        item.specOptions.reduce((sum, opt) => sum + opt.priceDelta, 0);
      total += itemPrice * item.quantity;
      count += item.quantity;
    });
    return { total: total.toFixed(2), count };
  },

  clearCart() {
    this.globalData.cart = [];
  },
});
```

```js
const app = getApp();

function request(method, path, data) {
  return new Promise((resolve, reject) => {
    wx.request({
      method,
      url: `${app.globalData.apiBase}${path}`,
      data,
      header: { 'Content-Type': 'application/json' },

        if (res.statusCode >= 200 && res.statusCode < 300) {
          resolve(res.data);
        } else {
          reject(res.data);
        }
      },
      fail: (err) => reject(err),
    });
  });
}
```
